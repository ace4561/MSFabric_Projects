--Gold Layer Olist FACT Tables Creation Script--

-- ============================================
-- Create schema if it doesn't already exist
-- ============================================

IF NOT EXISTS (SELECT 1 FROM sys.schemas WHERE name = 'Gold')
BEGIN
    EXEC('CREATE SCHEMA Gold');
END
GO

-- ============================================
-- FACT: fact_olist_payments
-- ============================================

CREATE TABLE Gold.fact_olist_payments (
    payment_sk               BIGINT IDENTITY NOT NULL,
    order_sk                 BIGINT NULL,
    customer_sk              BIGINT NULL,
    order_id                 VARCHAR(50),
    payment_sequential       INT NOT NULL,
    payment_type             VARCHAR(20),
    payment_installments     INT,
    payment_value            DECIMAL(10,2),
    created_at               DATETIME2(0) NOT NULL,
    updated_at               DATETIME2(0) NOT NULL
);
GO

ALTER TABLE Gold.fact_olist_payments
ADD CONSTRAINT PK_fact_payments PRIMARY KEY NONCLUSTERED (payment_sk) NOT ENFORCED;
GO
